cmake_minimum_required(VERSION 3.8)
project(gen2_sensors)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
include_directories(include)
add_executable(iahrs_node src/iahrs_node.cpp)
ament_target_dependencies(iahrs_node rclcpp sensor_msgs diagnostic_msgs)
add_executable(scan_monitor src/scan_monitor.cpp)
ament_target_dependencies(scan_monitor rclcpp sensor_msgs diagnostic_msgs)
install(TARGETS iahrs_node scan_monitor DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
ament_package()

cmake_minimum_required(VERSION 3.8)
project(gen2_control)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
include_directories(include)
add_library(wb_core STATIC src/wb_core.cpp)
set_target_properties(wb_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
find_package(rclcpp REQUIRED)
find_package(gen2_hardware REQUIRED)
find_package(gen2_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(gen2_sensors REQUIRED)
add_executable(balance_node src/balance_node.cpp)
target_link_libraries(balance_node wb_core gen2_hardware::gen2_hardware_core gen2_hardware::gen2_hardware_params)
ament_target_dependencies(balance_node rclcpp gen2_msgs sensor_msgs geometry_msgs std_msgs std_srvs gen2_sensors)
install(TARGETS balance_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/make_leg_table.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(TARGETS wb_core EXPORT export_${PROJECT_NAME} ARCHIVE DESTINATION lib)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_wb_core test/test_wb_core.cpp)
  target_link_libraries(test_wb_core wb_core)
  target_compile_definitions(test_wb_core PRIVATE GOLDEN_DIR="${CMAKE_CURRENT_SOURCE_DIR}/test")
endif()
ament_export_include_directories(include)
ament_export_libraries(wb_core)
ament_package()
